#### 1.2 Modificar método `_project_event()`

**Local:** Após linha 301 (após `logger.debug(f"Event {event.event_id} projected successfully")`)

**Adicionar:**

```python
# ETAPA 5: Project to Activity Stream (WebSocketEventLog)
try:
    ws_log_service = WebSocketEventLogService()
    sequence = ws_log_service.log_event(
        db=self.db_session,
        event_id=event.event_id,
        event_type=event.event_type.value,
        tenant_id=tenant_id,
        project_id=project_id,
        payload=event.payload,  # Full payload for Activity Stream
    )
    
    if sequence:
        logger.debug(f"Event {event.event_id} logged to Activity Stream (sequence={sequence})")
    else:
        logger.warning(f"Failed to log event {event.event_id} to Activity Stream (may be duplicate)")
        # Non-fatal: Activity Stream is optional, don't fail the main projection
except Exception as e:
    logger.error(f"Error logging event {event.event_id} to Activity Stream: {e}")
    # Non-fatal: Don't interrupt main projection pipeline
    # Activity Stream failure does not invalidate AgentEvent projection
```

**Lógica:**
- Idempotência: `log_event()` retorna `None` se for duplicata (mesma `event_id`)
- Não-fatal: Se falhar, apenas log warning (não interrompe projeção principal)
- Atomicidade: Na mesma transação PostgreSQL que AgentEvent
- Isolamento: Usa `tenant_id` validado já extraído

#### 1.3 Verificar que transação PostgreSQL é compartilhada

```python

```

**Verão após modificação:** `log_event()` usa `self.db_session` passado diretamente

**Atomicidade garantida:** Ambas AgentEvent e WebSocketEventLog são commitadas juntas ou roleback juntas

def test_event_projected_idempotent():
    """
    Verificar que processar o mesmo evento 2x não duplica Activity Stream
    """
    event = create_test_event(mission_id=test_mission_id)
    worker = ProjectionWorker(...)
    
    # First projection
    result1 = worker._project_event(event)
    assert result1 is True
    
    ws_count_1 = db.query(WebSocketEventLog).filter_by(event_id=event.event_id).count()
    assert ws_count_1 == 1
    
    # Second projection (simulating retry)
    result2 = worker._project_event(event)
    assert result2 is True
    
    ws_count_2 = db.query(WebSocketEventLog).filter_by(event_id=event.event_id).count()
    assert ws_count_2 == 1  # Still 1, not duplicated
```

### Teste 3: Multi-tenant isolation

def test_websocket_log_multi_tenant():
    """
    Verificar que eventos de tenant A não aparecem para tenant B
    """
    event_a = create_test_event(mission_id=mission_a, tenant_id=tenant_a_id)
    event_b = create_test_event(mission_id=mission_b, tenant_id=tenant_b_id)
    worker = ProjectionWorker(...)
    
    worker._project_event(event_a)
    worker._project_event(event_b)
    
    # Tenant A deve ver apenas seus eventos
    ws_events_a = db.query(WebSocketEventLog).filter_by(tenant_id=tenant_a_id).all()
    assert len(ws_events_a) == 1
    assert ws_events_a[0].event_id == event_a.event_id
    
    # Tenant B deve ver apenas seus eventos
    ws_events_b = db.query(WebSocketEventLog).filter_by(tenant_id=tenant_b_id).all()
    assert len(ws_events_b) == 1
    assert ws_events_b[0].event_id == event_b.event_id
```
